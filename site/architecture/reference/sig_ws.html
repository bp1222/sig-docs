sig_ws.h

include/sig/http/sig_ws.h

WebSocket (RFC 6455) handshake and framing primitives. Pure functions: no sockets, no I/O. Only unfragmented text and control frames are supported.

sig_ws_opcode

typedef enum { SIG_WS_TEXT = 1, SIG_WS_CLOSE = 8, SIG_WS_PING = 9, SIG_WS_PONG = 10 } sig_ws_opcode;

sig_ws_decode_result

typedef enum {
    SIG_WS_FRAME,          /* one complete frame decoded into *out */
    SIG_WS_INCOMPLETE,     /* need more bytes */
    SIG_WS_PROTOCOL_ERROR, /* unmasked client frame, reserved bits, unknown opcode,
                              fragmented (FIN clear or opcode 0), or control frame > 125 bytes */
    SIG_WS_TOO_LARGE       /* payload exceeds max_payload */
} sig_ws_decode_result;

sig_ws_frame

typedef struct {
    sig_ws_opcode op;
    unsigned char *payload; /* points into the caller's buffer, unmasked in place */
    size_t len;
    size_t consumed; /* bytes of the buffer this frame occupied; advance by this */
} sig_ws_frame;
